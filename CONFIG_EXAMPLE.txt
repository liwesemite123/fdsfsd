# Example:
#   customer1@example.com
#   customer2@example.com
#   customer3@example.com

# ========================================
# PROXIES (proxy/proxies.txt)
# ========================================
# One proxy per line
# Supported formats:
#   socks5://username:password@host:port
#   http://username:password@host:port
#   https://username:password@host:port
#
# Example:
#   socks5://user123:pass456@proxy.example.com:1080
#   http://proxyuser:proxypass@192.168.1.100:8080

# ========================================
# MESSAGE CUSTOMIZATION (in script)
# ========================================
# Edit carrd_automation.py and modify these constants:
#
# DEFAULT_NAME = "Your Name Here"
#   → The name that will appear in the form submission
#
# DEFAULT_EMAIL = "your.email@example.com"
#   → The email that will appear as the sender
#
# DEFAULT_MESSAGE = "Your message content here"
#   → The message that will be sent through the form
#
# Example:
#   DEFAULT_NAME = "John Smith"
#   DEFAULT_EMAIL = "john.smith@business.com"
#   DEFAULT_MESSAGE = "Hello! I'm interested in your services. Please contact me."
